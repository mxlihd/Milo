name: Build Android APK

on:
  workflow_dispatch:        # lets you press "Run workflow" by hand
  push:
    branches: [ main ]      # and it runs by itself every time you upload a change

jobs:
  apk:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-node@v4
        with:
          node-version: 20

      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: 17

      - name: Install tools
        run: npm install

      - name: Create the Android project
        run: npx cap add android

      - name: Apply MILO settings (landscape, fullscreen, icons)
        run: node scripts/patch-android.js

      - name: Copy the game into the app
        run: npx cap sync android

      - name: Build the APK
        working-directory: android
        run: |
          chmod +x gradlew
          ./gradlew assembleDebug

      - name: Save the APK so you can download it
        uses: actions/upload-artifact@v4
        with:
          name: MILO-apk
          path: android/app/build/outputs/apk/debug/app-debug.apk
